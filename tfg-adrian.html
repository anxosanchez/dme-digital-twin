graph LR %% — CONFIGURACIÓN DE ESTILOS QUÍMICOS — classDef styleEntradas fill:#1E3A8A,stroke:#1E3A8A,color:#fff,stroke-width:2px; classDef styleAparatos fill:#F3F4F6,stroke:#9CA3AF,color:#111827,stroke-width:2px; classDef styleLoxicas fill:#DBEAFE,stroke:#3B82F6,color:#1E40AF,stroke-width:1px,stroke-dasharray: 5 5; classDef styleProducto fill:#10B981,stroke:#059669,color:#fff,stroke-width:3px; classDef styleRefugallo fill:#EF4444,stroke:#DC2626,color:#fff,stroke-width:1px;

%% --- ENTRADAS ELEMENTAIS ---
InBiomasa([Biomasa: C, H, O, N, S, Ash]) --> R1-1
InAire([Aire: 21% O2 / 79% N2]) --> R1-3

%% --- MÓDULO I: BALANCES ELEMENTAIS E GASIFICACIÓN ---
subgraph Modulo_I [MODULO I: EVOLUCION QUIMICA DO SYNGAS]
    R1-1[Secadoiro R1-1] --> OpSecado{RStoic: Liberacion H2O}
    OpSecado --> R1-2[Reactor Pirolise R1-2]
    R1-2 --> OpPiro{RYield: Ruptura Macromolecular}
    
    OpPiro --> R1-3[Gasificador R1-3]
    R1-3 --> OpGibbs{RGibbs: Reducion de Char a CO <br/> C + CO2 -> 2CO}
    OpGibbs --> SEP1-1[Separador Solidos]
    
    SEP1-1 --> OpCinza{Inertes Solidos Out}
    OpCinza --> OutCinza[Refugallo: 100% Cinzas Sólidas]:::styleRefugallo
    
    OpCinza -- Syngas Bruto --> E1M[Tren Termico]
    E1M --> OpCool1{Arrefriamento}
    OpCool1 --> F1-1[Separador Flash F1-1]
    F1-1 --> OpFlash1{Condensacion de Auga}
    OpFlash1 --> OutAgua1[Auga Condensada: 99.8% H2O]:::styleRefugallo
end

%% Conexión inter-módulo (Syngas Seco Deseseñado)
OpFlash1 -- Composición Molar: <br/> H2: 55% | CO: 36% <br/> CO2: 5% | N2: 4% --> H2-1

%% --- MÓDULO II: LAZO DE SÍNTESE DE METANOL ---
subgraph Modulo_II [MODULO II: SINTESE DE CH3OH]
    H2-1[Quentador H2-1] --> OpHeat1{Syngas Gasoso}
    OpHeat1 --> K2-1[Compresor K2-1]
    K2-1 --> OpComp{Mestura con Reciclo I}
    OpComp --> R2-1[Reactor PFR R2-1]
    R2-1 --> OpLHHW{Cinetica LHHW: <br/> CO + 2H2 -> CH3OH <br/> CO2 + 3H2 -> CH3OH + H2O}
end

%% Saída do Reactor (Gases + MeOH producido)
OpLHHW --> V2-1

%% --- MÓDULO III: BALANCE DE INERTES (O TAPÓN DE N2) ---
subgraph Modulo_III [MODULO III: SEPARACION DE INERTES]
    V2-1[Valvula V2-1] --> OpExp1{Condensacion Parcial}
    OpExp1 --> F2-1[Separador Flash F2-1]
    
    %% LAZO CRÍTICO DO NITROXENO (418 kmol/h)
    F2-1 -- Fase Gas: Inertes Recirculantes --> SPT2-1[Divisor SPT2-1]
    SPT2-1 --> OpSplit1{Purga para evitar <br/> Envenenamento do Lazo}
    
    OpSplit1 -- Purga 25%: Saida N2 Neto --> OutPurga1[Gases de Purga: <br/> Equilibrio N2: 18.54% <br/> CO + H2 Arrastrados]:::styleRefugallo
    
    %% RETORNO DO LAZO I (N2 acumulado no SetPoint de Aspen)
    OpSplit1 -. Reciclo I: Gas Diluido <br/> N2 fixado en 18.54% .-> K2-1
    
    F2-1 -- Crudo de Metanol --> V2-2[Valvula V2-2]
    V2-2 --> OpExp2{Desgasificacion}
    OpExp2 --> S2-1[Columna Separadora S2-1]
    S2-1 --> OpMix1{Mestura con Retorno II <br/> Concentracion: 97.41% CH3OH}
end

%% Conexión inter-módulo (Alimentación Front-End DME)
OpMix1 --> P3-1

%% --- MÓDULO IV: DESHIDRATACIÓN A DME ---
subgraph Modulo_IV [MODULO IV: REACCION SINTESE DME]
    P3-1[Bomba P3-1] --> OpPump1{Fase Vapor}
    OpPump1 --> R3-1[Reactor Equilibrio R3-1]
    R3-1 --> OpEquil{REquil: <br/> 2 CH3OH -> DME + H2O}
end

%% Mestura Multicompoñente cara as torres (DME, CH3OH, H2O)
OpEquil --> D3-1

%% --- MÓDULO V: PURIFICACIÓN MESH DA MATRIZ ---
subgraph Modulo_V [MODULO V: SEPARACION MULTICOMPONENTE]
    D3-1[Torre Destilacion D3-1] --> OpMESH1{Modelo MESH: <br/> DME Separado por Volatilidade}
    OpMESH1 --> OutDME[[PRODUTO FINAL: <br/> 99.90% DME Puro <br/> H2O / CH3OH < 0.1%]]:::styleProducto
    
    OpMESH1 -- Colas: CH3OH + H2O --> V3-1[Valvula V3-1]
    V3-1 --> OpCond3{Alimentacion D3-2}
    OpCond3 --> D3-2[Torre Destilacion D3-2]
    D3-2 --> OpMESH2{Separacion Binaria <br/> Metanol / Auga}
    
    OpMESH2 -- Colas Fondo --> OutAgua2[Refugallo: Auga Limpa <br/> Pureza > 99.5% H2O]:::styleRefugallo
    OpMESH2 -- Cabezas: Metanol Recuperado --> P3-2[Bomba Reciclo P3-2]
    P3-2 --> OpPump2{Impulsion Retorno II}
    
    %% RETORNO DO LAZO II (Metanol recuperado pecha o balance)
    OpPump2 -. Lazo Reciclo II: <br/> 97.32% CH3OH Reutilizado .-> S2-1
end

%% --- ASIGNACIÓN DE ESTILOS ---
class InBiomasa,InAire styleEntradas;
class R1-1,R1-2,R1-3,SEP1-1,E1M,F1-1,H2-1,K2-1,R2-1,V2-1,F2-1,SPT2-1,V2-2,S2-1,P3-1,R3-1,D3-1,V3-1,D3-2,P3-2 styleAparatos;
class OpSecado,OpPiro,OpGibbs,OpCinza,OpCool1,OpFlash1,OpHeat1,OpComp,OpLHHW,OpExp1,OpSplit1,OpExp2,OpMix1,OpPump1,OpEquil,OpMESH1,OpCond3,OpMESH2,OpPump2 styleLoxicas;
